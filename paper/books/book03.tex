\section{Book III --- Circles}
\label{sec:book-III}

Book III introduces the geometry of the circle: tangents, chords,
inscribed angles, and the closely-related Thales' theorem (III.31).

\begin{claim}[Proposition III.1: Find the centre of a given circle]
\label{prop:III.1}
To find the centre of a given circle.
\end{claim}
\begin{evidence}[Proof sketch of III.1]
\label{ev:III.1}
Bisect a chord by I.10, erect a perpendicular by I.11, bisect the
perpendicular; the intersection is the centre.
\dependson{III.1}{I.10}
\dependson{III.1}{I.11}
\end{evidence}

\begin{claim}[Proposition III.31: Thales' theorem]
\label{prop:III.31}
In a circle the angle in the semicircle is right, that in a greater
segment less than a right angle, and that in a less segment greater
than a right angle.
\end{claim}
\begin{evidence}[Proof sketch of III.31]
\label{ev:III.31}
Apply I.5, I.32, I.13.
\dependson{III.31}{I.5}
\dependson{III.31}{I.13}
\dependson{III.31}{I.32}
\end{evidence}
